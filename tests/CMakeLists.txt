# The DLLs a test needs on Windows: Qt's, and OpenSSL's libcrypto (the cellular
# link) from wherever FindOpenSSL found it. ENVIRONMENT_MODIFICATION keeps the
# ';' of a Windows PATH, which ENVIRONMENT would split into list items.
find_package(OpenSSL 1.1.1 QUIET COMPONENTS Crypto)
set(CORV_TEST_PATH "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
if(WIN32 AND OPENSSL_INCLUDE_DIR)
    list(APPEND CORV_TEST_PATH "PATH=path_list_prepend:${OPENSSL_INCLUDE_DIR}/../bin")
endif()
if(WIN32 AND TARGET SDL3::SDL3)
    list(APPEND CORV_TEST_PATH "PATH=path_list_prepend:$<TARGET_FILE_DIR:SDL3::SDL3>")
endif()

function(corv_test name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE Qt6::Test ${ARGN})
    add_test(NAME ${name} COMMAND ${name})
    # A test that hangs fails after two minutes (CTest waits 25 by itself)
    set_tests_properties(${name} PROPERTIES ENVIRONMENT_MODIFICATION "${CORV_TEST_PATH}" TIMEOUT 120)
endfunction()
